//! Typed Binance request classification over one shared local quota owner.
#![forbid(unsafe_code)]
use tickwire_binance_clock as time;
use tickwire_binance_private as private;
use tickwire_binance_public as public;
use tickwire_quota as quota;
use tickwire_venue as venue;
pub use tickwire_clock::{ClockDomain, MonoTime, Source};
pub use quota::{Class, Consumed, Cost};
pub use tickwire_binance::request::Transport;
pub const MAX_RULES: usize=256;
pub const MAX_IDENTITIES: usize=4096;
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
pub enum Error { Invalid, Identity, Policy, Capacity, Quota(quota::Error) }
pub type Result<T>=core::result::Result<T,Error>;
impl From<quota::Error> for Error { fn from(e:quota::Error)->Self {Self::Quota(e)} }
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
pub enum CommandAction {
    New,Cancel,CancelReplace,Amend,Modify,Query,CancelList,Algo,CancelAlgo,
    SpotList {kind:venue::ListKind,legs:u8,advanced:bool},PeggedNew,AdvancedCancelReplace,Sor,
}
#[derive(Clone,Copy,Debug,PartialEq,Eq)]
pub enum Action {
    TimeConnect,TimeQuery,
    PrivateStreamConnect,PrivateManagementConnect,PrivateUpgrade,PrivatePong,PrivateClose,
    SpotLogon,SpotSubscribe,SpotSubscribeSignature,SpotStatus,SpotSubscriptions,SpotUnsubscribe,SpotLogout,
    UsdmCreate,UsdmKeepalive,UsdmClose,
    Public {lane:public::Lane,kind:public::ChargeKind},
    Command {transport:Transport,action:CommandAction},
}
#[derive(Clone,Debug,PartialEq,Eq)]
pub struct Rule {
    pub source:Source,
    pub action:Action,
    pub class:Class,
    pub api_weight:u32,
    pub websocket_messages:u32,
    pub costs:Vec<Cost>,
}
#[derive(Clone,Debug)]
pub struct Policy {
    pub generation:u64,
    pub starts_ns:u64,
    pub ends_ns:u64,
    pub rules:Vec<Rule>,
}
#[derive(Clone,Copy,Debug)]
pub struct TimeBinding {pub owner_id:[u8;16],pub source:Source,pub domain:ClockDomain}
#[derive(Clone,Debug)]
pub struct PrivateBinding {pub owner_id:[u8;16],pub scope:private::AccountScope,pub domain:ClockDomain}
#[derive(Clone,Copy,Debug)]
pub struct PublicBinding {pub owner_id:[u8;16],pub source:Source,pub domain:ClockDomain}
#[derive(Clone,Copy,Debug)]
pub struct CommandBinding {
    pub owner_id:[u8;16],pub risk_domain_id:[u8;16],pub account:u64,pub product:venue::Product,
    pub environment:venue::Environment,pub domain:ClockDomain,
}
pub struct Endpoint {limiter:quota::ExecLimiter}
impl Endpoint {
    pub fn new(limiter:quota::ExecLimiter,_producer:[u8;16],_domain:ClockDomain,_policy:Policy)->Result<Self> { Ok(Self{limiter}) }
    pub fn consume_time(&mut self,_binding:&TimeBinding,_charge:&time::TimeCharge,_now:MonoTime)->Result<Consumed> {Err(Error::Policy)}
    pub fn consume_private(&mut self,_binding:&PrivateBinding,_charge:&private::Charge,_now:MonoTime)->Result<Consumed> {Err(Error::Policy)}
    pub fn consume_public(&mut self,_binding:&PublicBinding,_charge:&public::PublicCharge,_now:MonoTime)->Result<Consumed> {Err(Error::Policy)}
    pub fn consume_command(&mut self,_binding:&CommandBinding,_command:&venue::CanonicalCommand,_transport:Transport,_now:MonoTime)->Result<Consumed> {Err(Error::Policy)}
    pub fn remaining(&self,key:quota::WindowKey,class:Class)->u64 {self.limiter.remaining(key,class)}
}
