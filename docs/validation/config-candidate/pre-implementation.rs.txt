//! Versioned configuration values. No process, file or trading authority.
#![forbid(unsafe_code)]

use serde_json::Value;
use sha2::{Digest, Sha256};

pub type Hash = [u8; 32];
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Format {
    Json,
    Toml,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Error {
    Encoding,
    Capacity,
    Scalar,
    Duplicate,
    Depth,
    Schema,
    Revision,
}
pub type Result<T> = core::result::Result<T, Error>;
#[derive(Debug, Clone, Copy)]
pub struct Limits {
    pub max_bytes: usize,
    pub max_depth: usize,
}
impl Default for Limits {
    fn default() -> Self {
        Self {
            max_bytes: 16 * 1024 * 1024,
            max_depth: 32,
        }
    }
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Candidate {
    value: Value,
    canonical: Vec<u8>,
    hash: Hash,
}
impl Candidate {
    pub fn parse(format: Format, bytes: &[u8], _limits: Limits) -> Result<Self> {
        let value = match format {
            Format::Json => serde_json::from_slice(bytes).map_err(|_| Error::Encoding)?,
            Format::Toml => {
                let input = std::str::from_utf8(bytes).map_err(|_| Error::Encoding)?;
                let table: toml::Table = input.parse().map_err(|_| Error::Encoding)?;
                serde_json::to_value(table).map_err(|_| Error::Encoding)?
            }
        };
        let canonical = serde_json::to_vec(&value).map_err(|_| Error::Encoding)?;
        let hash = Sha256::digest(&canonical).into();
        Ok(Self {
            value,
            canonical,
            hash,
        })
    }
    pub fn value(&self) -> &Value {
        &self.value
    }
    pub fn canonical_bytes(&self) -> &[u8] {
        &self.canonical
    }
    pub fn hash(&self) -> Hash {
        self.hash
    }
    pub fn validate_structural(&self) -> Result<()> {
        Ok(())
    }
}
