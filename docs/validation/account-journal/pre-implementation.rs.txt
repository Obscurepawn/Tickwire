//! Ordered account persistence; receipts are durability evidence, not trading authority.
#![forbid(unsafe_code)]
use std::{io, path::Path};
use tickwire_account::{Account, RecoveryRecord, Scope};
use tickwire_account_codec::{CanonicalBaseline, CanonicalRecord, Hash};
use tickwire_journal::{PhysicalFrontier, store::Store};

pub const MAX_RECORDS: usize = tickwire_account::MAX_FACTS + tickwire_account::MAX_ORDERS;
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub physical: tickwire_journal::Config,
    pub baseline_digest: Hash,
    pub initial_writer_epoch: u64,
    pub scope: Scope,
}
#[derive(Debug)]
pub enum Error {
    Io(io::Error),
    Physical(tickwire_journal::Error),
    Codec(tickwire_account_codec::Error),
    Account(tickwire_account::Error),
    Verification(VerificationError),
    Identity,
    Sequence,
    Epoch,
    AlreadyExists,
    Conflict,
    Capacity,
    Poisoned,
    Recovered,
}
pub type Result<T> = std::result::Result<T, Error>;
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerificationError {
    Missing,
    Invalid,
    Unavailable,
}
pub trait RecoveryVerifier {
    fn verify_admission(
        &mut self,
        baseline: &CanonicalBaseline,
        record: &CanonicalRecord,
    ) -> std::result::Result<(), VerificationError>;
}
/// Implementations synchronize immutable account files and honor Store guarantees.
pub trait Backend: Store {
    fn read_bootstrap(&mut self) -> io::Result<Option<Vec<u8>>>;
    fn publish_bootstrap(&mut self, bytes: &[u8]) -> io::Result<()>;
    fn read_baseline(&mut self) -> io::Result<Option<Vec<u8>>>;
    fn publish_baseline(&mut self, bytes: &[u8]) -> io::Result<()>;
}
pub struct FileBackend;
impl FileBackend {
    pub fn open(_path: impl AsRef<Path>) -> io::Result<Self> {
        Ok(Self)
    }
}
impl Store for FileBackend {
    fn read_manifest(&mut self) -> io::Result<Option<Vec<u8>>> { Ok(None) }
    fn publish_manifest(&mut self, _: &[u8]) -> io::Result<()> { Ok(()) }
    fn segment_ids(&mut self) -> io::Result<Vec<u64>> { Ok(vec![]) }
    fn read_segment(&mut self, _: u64, _: u64) -> io::Result<Vec<u8>> { Ok(vec![]) }
    fn create_segment(&mut self, _: u64) -> io::Result<()> { Ok(()) }
    fn append_segment(&mut self, _: u64, _: u64, _: &[u8]) -> io::Result<()> { Ok(()) }
    fn sync_segment(&mut self, _: u64) -> io::Result<()> { Ok(()) }
    fn truncate_segment(&mut self, _: u64, _: u64) -> io::Result<()> { Ok(()) }
}
impl Backend for FileBackend {
    fn read_bootstrap(&mut self) -> io::Result<Option<Vec<u8>>> { Ok(None) }
    fn publish_bootstrap(&mut self, _: &[u8]) -> io::Result<()> { Ok(()) }
    fn read_baseline(&mut self) -> io::Result<Option<Vec<u8>>> { Ok(None) }
    fn publish_baseline(&mut self, _: &[u8]) -> io::Result<()> { Ok(()) }
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct AppendReceipt {
    baseline: Hash,
    sequence: u64,
    digest: Hash,
    physical: PhysicalFrontier,
}
impl AppendReceipt {
    pub fn baseline_digest(&self) -> Hash { self.baseline }
    pub fn sequence(&self) -> u64 { self.sequence }
    pub fn record_digest(&self) -> Hash { self.digest }
    pub fn physical_frontier(&self) -> &PhysicalFrontier { &self.physical }
}
pub struct AccountJournal<S: Backend> {
    _backend: S,
}
pub struct Recovered<S: Backend> {
    pub journal: AccountJournal<S>,
    pub account: Account,
    pub discarded_tail_bytes: u64,
}
impl<S: Backend> AccountJournal<S> {
    pub fn create(_backend: S, _config: Config, _baseline: CanonicalBaseline) -> Result<Self> {
        Err(Error::Poisoned)
    }
    pub fn recover(
        _backend: S,
        _expected: Config,
        _new_writer_epoch: u64,
        _required: Option<&PhysicalFrontier>,
        _verifier: &mut impl RecoveryVerifier,
    ) -> Result<Recovered<S>> {
        Err(Error::Poisoned)
    }
    pub fn append(&mut self, _record: RecoveryRecord) -> Result<AppendReceipt> {
        Err(Error::Poisoned)
    }
    pub fn position(&self) -> u64 { 0 }
    pub fn writer_epoch(&self) -> u64 { 0 }
    pub fn poisoned(&self) -> bool { false }
    pub fn recovered(&self) -> bool { false }
    pub fn records(&self) -> &[CanonicalRecord] { &[] }
}
