struct Local {
    ns: Arc<AtomicU64>,
    generation: Arc<AtomicU64>,
}
impl clock::MonotonicClock for Local {
    fn sample(&mut self) -> clock::Result<clock::LocalTime> {
        Ok(clock::LocalTime {
            instant: Instant::now(),
            monotonic: MonoTime::new(
                ClockDomain::new([4; 16], self.generation.load(Ordering::SeqCst)).unwrap(),
                self.ns.load(Ordering::SeqCst),
            ),
        })
    }
}
fn config(source: Source) -> clock::Config {
    clock::Config {
        source,
        domain: ClockDomain::new([4; 16], 1).unwrap(),
        calibration: clock::CalibrationConfig {
            max_round_trip_ns: SECOND,
            max_width_ns: 2 * SECOND,
            max_age_ns: 10 * SECOND,
            quantization_error_ns: MILLISECOND,
            drift_ppb: 0,
            max_adjustment_ns: 10 * MILLISECOND,
        },
        unit: clock::TimeUnit::Milliseconds,
        first_sample_serial: 1,
        first_connection_epoch: 1,
        connect_timeout_ns: 5 * SECOND,
        request_timeout_ns: 2 * SECOND,
        max_response_bytes: 4096,
        max_headers: 16,
        io: clock::Limits::default(),
    }
}
fn certs() -> (
    Arc<clock::rustls::ClientConfig>,
    Arc<clock::rustls::ServerConfig>,
) {
    let cert = rcgen::generate_simple_self_signed(vec![
        "testnet.binance.vision".into(),
        "ws-api.testnet.binance.vision".into(),
    ])
    .unwrap();
    let server = clock::rustls::ServerConfig::builder()
        .with_no_client_auth()
        .with_single_cert(
            vec![cert.cert.der().clone()],
            clock::rustls::pki_types::PrivatePkcs8KeyDer::from(cert.signing_key.serialize_der())
                .into(),
        )
        .unwrap();
    let mut roots = clock::rustls::RootCertStore::empty();
    roots.add(cert.cert.der().clone()).unwrap();
    (
        Arc::new(
            clock::rustls::ClientConfig::builder()
                .with_root_certificates(roots)
                .with_no_client_auth(),
        ),
        Arc::new(server),
    )
}
type Fixture = (
    clock::BinanceTimeClient<Local>,
    Arc<AtomicU64>,
    Arc<AtomicU64>,
    Arc<clock::rustls::ClientConfig>,
    Arc<clock::rustls::ServerConfig>,
);
fn calibrated() -> Fixture {
    let (tls, server) = certs();
    let l = listener();
    let addr = l.local_addr().unwrap();
    let ns = Arc::new(AtomicU64::new(0));
    let generation = Arc::new(AtomicU64::new(1));
    let mut c = clock::BinanceTimeClient::new(
        config(Source::SpotTest),
        clock::NetworkConfig {
            address: addr,
            tls: tls.clone(),
        },
        Local {
            ns: ns.clone(),
            generation: generation.clone(),
        },
    )
    .unwrap();
    let s = server.clone();
    let h = thread::spawn(move || {
        let mut s = accept(&l, s);
        let h = headers(&mut s);
        assert!(h.starts_with("GET /api/v3/time HTTP/1.1\r\n"));
        assert!(h.contains("Host: testnet.binance.vision\r\n"));
        let b = br#"{"serverTime":1000}"#;
        write!(s, "HTTP/1.1 200 OK\r\nContent-Length: {}\r\n\r\n", b.len()).unwrap();
        s.write_all(b).unwrap();
        s.flush().unwrap();
    });
    let mut permit = |_: &clock::TimeCharge, _: &clock::LocalTime| true;
    c.start(&mut permit).unwrap();
    let end = Instant::now() + Duration::from_secs(3);
    loop {
        if let Some(o) = c
            .poll(&mut permit, clock::IoBudget::default())
            .unwrap()
            .outcome
        {
            assert!(o.calibration.is_ok());
            break;
        }
        assert!(Instant::now() < end);
        thread::yield_now();
    }
    h.join().unwrap();
    (c, ns, generation, tls, server)
}
fn budget(_: &AccountScope, _: &SignedTime, _: &ClockSample) -> Result<u64> {
    Ok(MILLISECOND)
}
#[test]
fn actual_https_calibration_drives_actual_signed_wss_subscription() {
    let (mut time, _, _, tls, server) = calibrated();
    let l = listener();
    let addr = l.local_addr().unwrap();
    let h = thread::spawn(move || {
        let mut s = accept(&l, server);
        upgrade(&mut s, "/ws-api/v3");
        let q = json(&mut s);
        verify_signature(&q, 0);
        assert_eq!(q["params"]["timestamp"], 1000);
        reply(&mut s, &q["id"], serde_json::json!({"subscriptionId":0}));
        let mut b = [0];
        let _ = s.read(&mut b);
    });
    let key = private::SigningKey::hmac(SECRET).unwrap();
    let mut client = private::PrivateClient::new(
        private::SessionConfig::new(scope(private::Market::Spot)),
        private::Authentication::SpotSignature {
            api_key: KEY,
            signing_key: &key,
        },
        private::NetworkConfig {
            stream_address: addr,
            management_address: None,
            tls,
        },
        5,
    )
    .unwrap();
    let mut budget = budget;
    let mut adapter =
        ClockAdapter::new(&mut time, scope(private::Market::Spot), &mut budget).unwrap();
    let mut permit = |_: &private::Charge, _: &ClockSample| true;
    let end = Instant::now() + Duration::from_secs(3);
    while client.status().phase != private::Phase::Receiving && Instant::now() < end {
        client
            .poll(
                &mut adapter,
                private::PollBudget::default(),
                &mut permit,
                |_| Ok(()),
            )
            .unwrap();
        thread::yield_now();
    }
    assert_eq!(client.status().phase, private::Phase::Receiving);
    client.stop(private::StopMode::Disconnect).unwrap();
    drop(client);
    h.join().unwrap();
}

#[test]
fn constructor_rejects_product_environment_and_invalid_account() {
    let (mut time, _, _, _, _) = calibrated();
    let mut b = budget;
    for mut s in [scope(private::Market::Usdm), scope(private::Market::Spot)] {
        if s.market == private::Market::Spot {
            s.environment = private::Environment::Production;
        }
        assert!(ClockAdapter::new(&mut time, s, &mut b).is_err());
    }
    let mut s = scope(private::Market::Spot);
    s.account = 0;
    assert!(ClockAdapter::new(&mut time, s, &mut b).is_err());
}
#[test]
fn original_scope_and_window_conversion_are_checked() {
    let (mut time, _, _, _, _) = calibrated();
    let reading = time.sample().unwrap();
    let now = ClockSample {
        instant: reading.local.instant,
        monotonic: reading.local.monotonic,
        signing_time: reading.signed,
    };
    let mut b = budget;
    let s = scope(private::Market::Spot);
    let mut adapter = ClockAdapter::new(&mut time, s.clone(), &mut b).unwrap();
    let mut other = s.clone();
    other.account += 1;
    assert!(
        adapter
            .check_signed(&other, now.signing_time.as_ref().unwrap(), &now, 5_000_000)
            .is_err()
    );
    assert!(
        adapter
            .check_signed(&s, now.signing_time.as_ref().unwrap(), &now, u64::MAX)
            .is_err()
    );
}

#[test]
fn new_sample_does_not_resign_the_original_stamp() {
    let (mut time, ns, _, _, _) = calibrated();
    let initial = time.sample().unwrap();
    let old = initial.signed.unwrap();
    let now = ClockSample {
        instant: initial.local.instant,
        monotonic: initial.local.monotonic,
        signing_time: Some(old),
    };
    ns.store(MILLISECOND, Ordering::SeqCst);
    let mut called = false;
    let mut b = |_: &AccountScope, stamp: &SignedTime, fresh: &ClockSample| {
        called = true;
        assert_eq!(stamp.timestamp(), 1000);
        assert_eq!(fresh.signing_time.unwrap().timestamp(), 1001);
        Ok(MILLISECOND)
    };
    let s = scope(private::Market::Spot);
    let mut adapter = ClockAdapter::new(&mut time, s.clone(), &mut b).unwrap();
    adapter.check_signed(&s, &old, &now, 5_000_000).unwrap();
    assert!(called);
}

    for change_domain in [false, true] {

            instant: r.local.instant,

            signing_time: r.signed,
        };
        let mut b = move |_: &AccountScope, _: &SignedTime, _: &ClockSample| {
            if change_domain {

        };

        let mut adapter = ClockAdapter::new(&mut time, s.clone(), &mut b).unwrap();
        assert_eq!(
            adapter.check_signed(&s, now.signing_time.as_ref().unwrap(), &now, 5_000_000),
            Err(if change_domain {
                Error::ClockDomain
            } else {
                Error::Clock
            })
        );
    }
}
#[test]
fn original_stamp_cannot_cross_product_or_clock_domain() {
    let (mut time, _, _, _, _) = calibrated();
    let r = time.sample().unwrap();
    let now = ClockSample {
        instant: r.local.instant,
        monotonic: r.local.monotonic,
        signing_time: r.signed,
    };
    let s = scope(private::Market::Spot);
    let mut b = budget;
    let mut adapter = ClockAdapter::new(&mut time, s.clone(), &mut b).unwrap();
    for wrong_domain in [false, true] {
        let domain = if wrong_domain {
            ClockDomain::new([4; 16], 2).unwrap()
        } else {
            now.monotonic.domain()
        };
        let mut c = tickwire_clock::Calibrator::new(
            config(Source::SpotTest).calibration,
            if wrong_domain {
                Source::SpotTest
            } else {
                Source::UsdmTest
            },
            domain,
            1,
        )
        .unwrap();
        let n = MonoTime::new(domain, 0);
        let token = c.begin(n).unwrap();
        c.finish(&token, n, 1000).unwrap();
        let stamp = c.sign_time(n, clock::TimeUnit::Milliseconds).unwrap();
        assert_eq!(
            adapter.check_signed(&s, &stamp, &now, 5_000_000),
            Err(if wrong_domain {
                Error::ClockDomain
            } else {
                Error::Identity
            })
        );
    }
}

fn mono_deadline_during_budget(advance_on_call: usize) -> private::ControlState {
    let (mut time, ns, _, tls, server) = calibrated();
    let l = listener();
    let addr = l.local_addr().unwrap();
    let peer = thread::spawn(move || {
        let mut s = accept(&l, server);
        upgrade(&mut s, "/ws-api/v3");
        let mut received = [0; 4096];
        match s.read(&mut received) {
            Ok(n) => n,
            Err(e) if e.kind() == std::io::ErrorKind::UnexpectedEof => 0,
            Err(e) => panic!("unexpected local server error: {e}"),
        }
    });
    let mut cfg = private::SessionConfig::new(scope(private::Market::Spot));
    cfg.timing.request_ns = SECOND;
    assert_eq!(cfg.timing.recv_window_micros, 5_000_000);
    let key = private::SigningKey::hmac(SECRET).unwrap();
    let mut client = private::PrivateClient::new(
        cfg,
        private::Authentication::SpotSignature {
            api_key: KEY,
            signing_key: &key,
        },
        private::NetworkConfig {
            stream_address: addr,
            management_address: None,
            tls,
        },
        5,
    )
    .unwrap();
    let mut calls = 0;
    let mut budget = |_: &AccountScope, _: &SignedTime, _: &ClockSample| {
        calls += 1;
        if calls == advance_on_call {
            // Suspend-aware MonoTime can advance while transport Instant does not.
            // The calibration and signed recvWindow remain valid at this time.
            ns.store(2 * SECOND, Ordering::SeqCst);
        }
        Ok(MILLISECOND)
    };
    let mut adapter =
        ClockAdapter::new(&mut time, scope(private::Market::Spot), &mut budget).unwrap();
    let mut permit = |_: &private::Charge, _: &ClockSample| true;
    let end = Instant::now() + Duration::from_secs(3);
    let mut observed = None;
    while observed.is_none() && Instant::now() < end {
        let _ = client.poll(
            &mut adapter,
            private::PollBudget::default(),
            &mut permit,
            |delivery| {
                if let private::Delivery::Control(outcome) = delivery {
                    observed = Some(outcome.state);
                }
                Ok(())
            },
        );
        thread::yield_now();
    }
    client.stop(private::StopMode::Disconnect).unwrap();
    drop(client);
    let received = peer.join().unwrap();
    drop(adapter);
    assert_eq!(ns.load(Ordering::SeqCst), 2 * SECOND);
    assert!(calls >= advance_on_call);
    // This confirms the candidate did not merely overrun the wall-clock test.
    assert!(Instant::now() < end);
    let state = observed.expect("expired request must publish a control outcome");
    eprintln!(
        "budget call {advance_on_call}: mono=2s, request deadline=1s, \
         calibration expiry=10s, peer decrypted bytes={received}, outcome={state:?}"
    );
    state
}

#[test]
fn mono_expiry_during_first_budget_check_rejects_before_acceptance() {
    assert_eq!(
        mono_deadline_during_budget(1),
        private::ControlState::NoBytes
    );
}

#[test]
fn mono_expiry_during_second_budget_check_rejects_before_acceptance() {
    assert_eq!(
        mono_deadline_during_budget(2),
        private::ControlState::NoBytes
    );
}
