use crate::*;
use serde::Deserialize;
use std::time::Duration;
use tickwire_clock::{Calibrator, MILLISECOND, SampleToken};
use tickwire_transport::{Connection, PreparedSend, Security, SendState, http};

struct Active {
    epoch: u64,
    connection: Connection,
    decoder: http::ResponseDecoder,
    token: Option<SampleToken>,
    queued: bool,
    deadline_ns: u64,
    deadline: Instant,
}
pub struct BinanceTimeClient<C: MonotonicClock> {
    config: Config,
    network: NetworkConfig,
    clock: C,
    calibrator: Calibrator,
    next_epoch: Option<u64>,
    last_instant: Option<Instant>,
    fatal: Option<Error>,
    active: Option<Active>,
}
impl<C: MonotonicClock> BinanceTimeClient<C> {
    pub fn source(&self) -> Source {
        self.config.source
    }
    pub fn domain(&self) -> ClockDomain {
        self.config.domain
    }
    pub fn new(config: Config, network: NetworkConfig, clock: C) -> Result<Self> {
        if config.first_connection_epoch == 0
            || config.connect_timeout_ns == 0
            || config.request_timeout_ns == 0
            || config.max_response_bytes == 0
            || config.max_response_bytes > 16 << 20
            || config.max_headers == 0
            || config.max_headers > 256
            || config.calibration.quantization_error_ns < MILLISECOND
            || network.tls.enable_early_data
            || (matches!(config.source, Source::UsdmProduction | Source::UsdmTest)
                && config.unit == TimeUnit::Microseconds)
            || [
                config.io.max_send_bytes,
                config.io.max_receive_bytes,
                config.io.tls_buffer_bytes,
                config.io.max_accept_bytes,
            ]
            .iter()
            .any(|&n| n == 0 || n > 16 << 20)
        {
            return Err(Error::Invalid);
        }
        let calibrator = Calibrator::new(
            config.calibration,
            config.source,
            config.domain,
            config.first_sample_serial,
        )
        .map_err(Error::Clock)?;
        Ok(Self {
            next_epoch: Some(config.first_connection_epoch),
            config,
            network,
            clock,
            calibrator,
            last_instant: None,
            fatal: None,
            active: None,
        })
    }
    fn local(&mut self) -> Result<LocalTime> {
        if let Some(error) = self.fatal {
            return Err(error);
        }
        let now = self.clock.sample().inspect_err(|error| {
            if matches!(
                error,
                Error::Clock(
                    tickwire_clock::Error::Domain
                        | tickwire_clock::Error::TimeBackwards
                        | tickwire_clock::Error::Overflow
                        | tickwire_clock::Error::ClockJump
                        | tickwire_clock::Error::Faulted
                )
            ) {
                self.fatal = Some(*error);
            }
        })?;
        if self.last_instant.is_some_and(|last| now.instant < last) {
            let e = Error::Clock(tickwire_clock::Error::TimeBackwards);
            self.fatal = Some(e);
            return Err(e);
        }
        self.last_instant = Some(now.instant);
        // Even before calibration, every local sample belongs to the one domain.
        match self.calibrator.read(now.monotonic) {
            Ok(_)
            | Err(
                tickwire_clock::Error::Uncalibrated
                | tickwire_clock::Error::Expired
                | tickwire_clock::Error::Uncertain,
            ) => Ok(now),
            Err(e) => Err(Error::Clock(e)),
        }
    }
    fn deadline(&mut self, now: LocalTime, span: u64) -> Result<(u64, Instant)> {
        match (
            now.monotonic.nanos().checked_add(span),
            now.instant.checked_add(Duration::from_nanos(span)),
        ) {
            (Some(ns), Some(instant)) => Ok((ns, instant)),
            _ => {
                let error = Error::Clock(tickwire_clock::Error::Overflow);
                self.fatal = Some(error);
                Err(error)
            }
        }
    }
    pub fn start(&mut self, permit: &mut impl TimePermit) -> Result<u64> {
        if self.active.is_some() {
            return Err(Error::Busy);
        }
        let now = self.local()?;
        let epoch = self.next_epoch.ok_or(Error::Exhausted)?;
        self.next_epoch = epoch.checked_add(1);
        if !permit.allow(
            &TimeCharge {
                source: self.config.source,
                domain: self.config.domain,
                connection_epoch: epoch,
                kind: ChargeKind::Connect,
                api_weight: 0,
            },
            &now,
        ) {
            return Err(Error::Denied);
        }
        let now = self.local()?;
        let (deadline_ns, deadline) = self.deadline(now, self.config.connect_timeout_ns)?;
        let name =
            rustls::pki_types::ServerName::try_from(authority(self.config.source).to_owned())
                .map_err(|_| Error::Invalid)?;
        let connection = Connection::connect(
            self.network.address,
            Security::Tls {
                server_name: name,
                config: self.network.tls.clone(),
            },
            self.config.io,
            deadline,
        )
        .map_err(|_| Error::Transport)?;
        self.active = Some(Active {
            epoch,
            connection,
            decoder: http::ResponseDecoder::new(
                self.config.max_response_bytes,
                self.config.max_headers,
            )
            .map_err(|_| Error::Invalid)?,
            token: None,
            queued: false,
            deadline_ns,
            deadline,
        });
        Ok(epoch)
    }
    fn done(
        &mut self,
        active: Active,
        http: Option<ResponseMetadata>,
        calibration: Result<Reading>,
    ) -> Outcome {
        if let Some(token) = active.token {
            let _ = self.calibrator.cancel(&token);
        }
        // Dropping this exact Connection prevents late responses from another sample.
        Outcome {
            connection_epoch: active.epoch,
            http,
            calibration,
        }
    }
    pub fn poll(&mut self, permit: &mut impl TimePermit, budget: IoBudget) -> Result<PollReport> {
        if budget.max_syscalls == 0 || budget.max_bytes == 0 {
            return Err(Error::Budget);
        }
        let mut active = self.active.take().ok_or(Error::Closed)?;
        let mut report = PollReport::default();
        let outcome = self.poll_active(&mut active, permit, budget, &mut report);
        match outcome {
            Ok(None) => self.active = Some(active),
            Ok(Some((meta, result))) => report.outcome = Some(self.done(active, meta, result)),
            Err(error) => report.outcome = Some(self.done(active, None, Err(error))),
        }
        Ok(report)
    }
    fn poll_active(
        &mut self,
        active: &mut Active,
        permit: &mut impl TimePermit,
        budget: IoBudget,
        report: &mut PollReport,
    ) -> Result<Option<(Option<ResponseMetadata>, Result<Reading>)>> {
        let now = self.local()?;
        if now.monotonic.nanos() >= active.deadline_ns || now.instant >= active.deadline {
            return Err(Error::Timeout);
        }
        if active.connection.is_ready() && !active.queued {
            let (deadline_ns, deadline) = self.deadline(now, self.config.request_timeout_ns)?;
            let bytes = http::encode_request(
                "GET",
                path(self.config.source),
                &[
                    ("Host", authority(self.config.source)),
                    ("Connection", "close"),
                ],
                &[],
                self.config.io.max_send_bytes,
            )
            .map_err(|_| Error::Invalid)?;
            active
                .connection
                .queue(
                    PreparedSend::new(&bytes, deadline, self.config.io.max_send_bytes)
                        .map_err(|_| Error::Invalid)?,
                )
                .map_err(|_| Error::Transport)?;
            active.queued = true;
            active.deadline = deadline;
            active.deadline_ns = deadline_ns;
        }
        if active.queued {
            let mut denied = None;
            let send = active
                .connection
                .try_send(now.instant, || {
                    let result = (|| {
                        let before = self.local()?;
                        if before.monotonic.nanos() >= active.deadline_ns
                            || before.instant >= active.deadline
                        {
                            return Err(Error::Timeout);
                        }
                        if !permit.allow(
                            &TimeCharge {
                                source: self.config.source,
                                domain: self.config.domain,
                                connection_epoch: active.epoch,
                                kind: ChargeKind::Query,
                                api_weight: 1,
                            },
                            &before,
                        ) {
                            return Err(Error::Denied);
                        }
                        let first = self.local()?;
                        if first.monotonic.nanos() >= active.deadline_ns
                            || first.instant >= active.deadline
                        {
                            return Err(Error::Timeout);
                        }
                        if active.token.is_none() {
                            active.token = Some(
                                self.calibrator
                                    .begin(first.monotonic)
                                    .map_err(Error::Clock)?,
                            );
                        }
                        Ok(())
                    })();
                    if let Err(error) = result {
                        denied = Some(error);
                        false
                    } else {
                        true
                    }
                })
                .map_err(|_| Error::Transport)?;
            if let Some(error) = denied {
                return Err(error);
            }
            if matches!(send.state, SendState::NoBytes(_) | SendState::Failed(_)) {
                return Err(Error::Transport);
            }
        }
        let drive = active.connection.drive(now.instant, budget);
        let ended = match &drive {
            Ok(d) => {
                report.syscalls = d.syscalls;
                report.network_bytes = d.network_read + d.network_written;
                d.peer_closed
            }
            Err(_) => {
                report.io_budget_exhausted = true;
                true
            }
        };
        // Drain all already-decrypted bytes before classifying a terminal socket.
        // The connection and decoder each have fixed, configured capacity bounds.
        let mut buffer = [0; 2048];
        loop {
            let n = active.connection.read(&mut buffer);
            if n == 0 {
                break;
            }
            active
                .decoder
                .push(&buffer[..n])
                .map_err(|_| Error::Protocol)?;
        }
        if ended {
            active.decoder.finish();
        }
        let response = active
            .decoder
            .next_response()
            .map_err(|_| Error::Protocol)?;
        if let Some(response) = response {
            if !active
                .connection
                .send_report()
                .is_some_and(|r| r.accepted_bytes == r.total_bytes)
                || !active.decoder.take_buffered().is_empty()
            {
                return Err(Error::Protocol);
            }
            let meta = ResponseMetadata {
                status: response.status,
                headers: response.headers,
            };
            if response.status != 200 {
                return Ok(Some((Some(meta), Err(Error::HttpStatus(response.status)))));
            }
            #[derive(Deserialize)]
            #[serde(deny_unknown_fields)]
            struct Body {
                #[serde(rename = "serverTime")]
                server_time: u64,
            }
            let body: Body = match serde_json::from_slice(&response.body) {
                Ok(body) => body,
                Err(_) => return Ok(Some((Some(meta), Err(Error::Protocol)))),
            };
            let received = match self.local() {
                Ok(now) => now,
                Err(error) => return Ok(Some((Some(meta), Err(error)))),
            };
            if received.monotonic.nanos() >= active.deadline_ns
                || received.instant >= active.deadline
            {
                return Ok(Some((Some(meta), Err(Error::Timeout))));
            }
            let token = active.token.as_ref().ok_or(Error::Protocol)?;
            let calibrated = self
                .calibrator
                .finish(token, received.monotonic, body.server_time)
                .map_err(Error::Clock);
            return Ok(Some((Some(meta), calibrated)));
        }
        if ended {
            return Err(if drive.is_err() {
                Error::Transport
            } else {
                Error::Closed
            });
        }
        Ok(None)
    }
    pub fn cancel(&mut self) -> Result<()> {
        let active = self.active.take().ok_or(Error::Closed)?;
        self.done(active, None, Err(Error::Closed));
        Ok(())
    }
    pub fn sample(&mut self) -> Result<ClockReading> {
        let local = self.local()?;
        let signed = match self.calibrator.sign_time(local.monotonic, self.config.unit) {
            Ok(signed) => Some(signed),
            Err(
                tickwire_clock::Error::Uncalibrated
                | tickwire_clock::Error::Expired
                | tickwire_clock::Error::Uncertain,
            ) => None,
            Err(e) => return Err(Error::Clock(e)),
        };
        Ok(ClockReading { local, signed })
    }
    pub fn check_send(
        &mut self,
        signed: &SignedTime,
        recv_window_ns: u64,
        budget_ns: u64,
    ) -> Result<()> {
        let now = self.local()?;
        self.calibrator
            .check_send(signed, now.monotonic, recv_window_ns, budget_ns)
            .map_err(Error::Clock)
    }
    pub fn status(&self) -> CalibrationStatus {
        if let Some(Error::Clock(error)) = self.fatal {
            CalibrationStatus::Faulted(error)
        } else {
            self.calibrator.status()
        }
    }
    pub fn in_flight(&self) -> Option<SampleProgress> {
        self.active.as_ref().map(|active| SampleProgress {
            connection_epoch: active.epoch,
            measurement_started: active.token.is_some(),
            accepted_application_bytes: active
                .connection
                .send_report()
                .map_or(0, |report| report.accepted_bytes),
        })
    }
    pub fn register(&mut self, registry: &mio::Registry, token: mio::Token) -> Result<()> {
        self.active
            .as_mut()
            .ok_or(Error::Closed)?
            .connection
            .register(registry, token)
            .map_err(|_| Error::Transport)
    }
    pub fn reregister(&mut self, registry: &mio::Registry, token: mio::Token) -> Result<()> {
        self.active
            .as_mut()
            .ok_or(Error::Closed)?
            .connection
            .reregister(registry, token)
            .map_err(|_| Error::Transport)
    }
}
fn authority(source: Source) -> &'static str {
    match source {
        Source::SpotProduction => "api.binance.com",
        Source::SpotTest => "testnet.binance.vision",
        Source::UsdmProduction => "fapi.binance.com",
        Source::UsdmTest => "demo-fapi.binance.com",
    }
}
fn path(source: Source) -> &'static str {
    match source {
        Source::SpotProduction | Source::SpotTest => "/api/v3/time",
        Source::UsdmProduction | Source::UsdmTest => "/fapi/v1/time",
    }
}
